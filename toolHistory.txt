323	char* read_line(FILE* stream)
324	{
325	
326	    int bufferSize = INITIAL_BUFFER_SIZE;
327	    char* buffer = malloc(sizeof(char) * bufferSize);
328	    int numRead = 0;
329	    int next;
330	
331	    if (feof(stream)) {
332	        return NULL;
333	    }
334	
335	    while (1) {
336	        next = fgetc(stream);
337	
338	        if (next == EOF && numRead == 0) {
339	            free(buffer);
340	            return NULL;
341	        }
342	
343	        if (numRead == bufferSize - 1) {
344	            bufferSize *= 2;
345	            buffer = realloc(buffer, sizeof(char) * bufferSize);
346	        }
347	
348	        if (next == '\n' || next == EOF) {
349	            buffer[numRead] = '\0';
350	            break;
351	        }
352	
353	        buffer[numRead++] = next;
354	    }
355	    return buffer;
356	}

358	/**
359	 * Separates segments of string separated by delim into array of char*
360	 **/
361	char** separate_line(char* string, char delim)
362	{
363	    char** tokens = (char**)malloc(sizeof(char*));
364	    tokens[0] = NULL;
365	
366	    char delimiters[] = {delim, '\0'};
367	
368	    int numTokens = 0;
369	    char* token = strtok(string, delimiters);
370	
371	    while (token) {
372	        tokens[numTokens] = strdup(token);
373	        numTokens++;
374	        tokens = (char**)realloc(
375	                (void*)tokens, sizeof(char*) * (numTokens + 1));
376	        tokens[numTokens] = NULL;
377	
378	        token = strtok(NULL, delimiters);
379	    }
380	
381	    return tokens;
382	}

384	te_variable* struct_to_te_var(Var** variables, Loop** loops,
385	        const int* variablesCount, const int* loopsCount)
386	{
387	    int varCountChecked;
388	    if (variablesCount != NULL) {
389	        varCountChecked = (*variablesCount);
390	    } else {
391	        varCountChecked = 0;
392	    }
393	
394	    int loopCountChecked;
395	    if (loopsCount != NULL) {
396	        loopCountChecked = (*loopsCount);
397	    } else {
398	        loopCountChecked = 0;
399	    }
400	
401	    int total = varCountChecked + loopCountChecked;
402	    te_variable* result = (te_variable*)malloc(sizeof(te_variable) * (total));
403	    for (int i = 0; i < varCountChecked; i++) {
404	        result[i].name = (*variables)[i].name;
405	        result[i].address = &((*variables)[i].value);
406	    }
407	
408	    for (int i = 0; i < loopCountChecked; i++) {
409	        result[varCountChecked + i].name = (*loops)[i].name;
410	        result[varCountChecked + i].address = &((*loops)[i].value);
411	    }
412	
413	    return result;
414	}
